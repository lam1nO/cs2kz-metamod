"Phrases"

	// ВНИМАНИЕ: у фразы С аргументами блок "#format" ОБЯЗАТЕЛЕН, а плейсхолдеры — токены
	// {name}, не printf-спецификаторы. Без "#format" PrepareMessageWithLang возвращает строку
	// как есть, НЕ вызывая tfm::format (kz_language.h:122-138): аргумент молча выбрасывается,
	// а printf-спецификатор доезжает до второго vsnprintf в KZPlayer::PrintChat уже без
	// varargs — то есть читается мусор из стека. Именно так тут и было до 17.08: весь файл
	// жил на `%s`/`%i` без "#format", и игрок вместо названия карты видел «null» или случайный
	// символ (жалоба 17.08). Цветовые токены {darkred}/{default} к "#format" отношения не
	// имеют — BuildTemplate заменяет только объявленные имена, остальные фигурные скобки
	// достаются CFormat во втором проходе.
	"Mcustom - Retry"
	{
		"#format"	"attempt:d,total:d"
		"en"	"Still downloading (attempt {attempt} of {total})…"
		"ru"	"Всё ещё качаю (попытка {attempt} из {total})…"
	}
	// Здесь подставляется workshop-ID намеренно: это аргумент команды, которую игроку
	// предлагают повторить — по названию карты !mcustom не сработает.
	"Mcustom - Exhausted"
	{
		"#format"	"id:s"
		"en"	"Failed to download the map — Steam connection issue. Try {darkred}!mcustom {id}{default} again in a minute."

	}

	}
